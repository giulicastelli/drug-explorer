empty_check.py:
    # Uses the v1 clean() of the current loader image: how many usable rows have an empty-string value?
    import collections, zipfile, ijson, load_fda as L
    counts = collections.Counter(); empty = collections.Counter(); n = 0
    for path in sorted(L.DATA_DIR.glob("*.json.zip")):
        with zipfile.ZipFile(path) as z, z.open(z.infolist()[0]) as f:
            for r in ijson.items(f, "results.item"):
                row = L.clean(r, {0, 1, 2, 3}, counts)
                if row is None: continue
                n += 1
                for c, v in zip(L.COLUMNS, row):
                    if v == "": empty[c] += 1
    print("usable rows", n, "empty-string values per column", dict(empty))
